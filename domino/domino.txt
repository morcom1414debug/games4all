        }

                enterLobby();
            } catch (err) { console.error(err); document.getElementById('btn-create-room').disabled = false; }
        };

        function joinRoom(roomId, hostPeerId) {
            isHost = false; currentRoomId = roomId; announce('กำลังเชื่อมต่อ...');
            hostConnection = peer.connect(hostPeerId, { reliable: true });
            hostConnection.on('open', () => { hostConnection.send({ type: 'joinReq', peerId: myPeerId, name: "ผู้เล่น" }); enterLobby(); hostConnection.on('data', handleClientData); });
        }
        function leaveLobby() { stopBGM(); if (isHost && currentRoomId) remove(ref(db, `domino_rooms/${currentRoomId}`)); location.reload(); }

        function renderLobby() {
            const list = document.getElementById('lobby-player-list'); list.innerHTML = '';
            document.getElementById('lobby-player-count').textContent = players.length;
            players.forEach(p => {
                const li = document.createElement('li'); li.style.padding = '10px'; li.style.background = 'rgba(0,0,0,0.3)'; li.style.marginBottom = '8px'; li.style.borderRadius = '8px';
                li.innerHTML = `<span>${p.isBot?'🤖':'👤'} ${p.name}</span>`;
                list.appendChild(li);
            });
            if(isHost) {
                document.getElementById('bot-count-display').textContent = players.filter(p=>p.isBot).length;
                document.getElementById('btn-add-bot').disabled = players.length >= MAX_PLAYERS;

            }
        }

        function setupHostConnection(conn) {
            conn.on('data', data => {
                if (data.type === 'joinReq' && players.length < MAX_PLAYERS) {
                    conn.customPeerId = data.peerId;
                    players.push({ id: data.peerId, name: `ผู้เล่น${players.length+1}`, isBot: false, score: 0, conn });
                    syncLobby(); update(ref(db, `domino_rooms/${currentRoomId}`), { currentPlayers: players.length });
                } else if (data.type === 'action') handlePlayerAction(conn.customPeerId || conn.peer, data.action, data.payload);
            });
        }

        document.getElementById('btn-add-bot').onclick = () => {
            if (players.length < MAX_PLAYERS) {
                const botName = BOT_NAMES[players.filter(p=>p.isBot).length % BOT_NAMES.length];
                players.push({ id: 'bot_'+Date.now(), name: botName, isBot: true, score: 0 });
                syncLobby(); playSound('select');
            }
        };
        document.getElementById('btn-remove-bot').onclick = () => {
            const botIdx = players.slice().reverse().findIndex(p => p.isBot);
            if (botIdx !== -1) { players.splice(players.length - 1 - botIdx, 1); syncLobby(); playSound('select'); }
        };

        function syncLobby() {
            if(!isHost) return;
            renderLobby();
            const safePlayers = players.map(p => ({ id: p.id, name: p.name, isBot: p.isBot, score: p.score }));
            connections.forEach(c => { if(c.open) c.send({ type: 'lobbySync', players: safePlayers }); });
        }

        function broadcastGameState() {
            if (!isHost) return;
            const safePlayers = players.map(p => ({ id: p.id, name: p.name, isBot: p.isBot, score: p.score }));
            const safeGame = { status: game.status, deckCount: game.deck.length, leftEnd: game.leftEnd, rightEnd: game.rightEnd, turnIndex: game.turnIndex, direction: game.direction, players: safePlayers, playerStates: {}, log: game.log };
            players.forEach(p => safeGame.playerStates[p.id] = { cardCount: game.playerStates[p.id].hand.length, declared: game.playerStates[p.id].declared });
            connections.forEach(c => {
                const tid = c.customPeerId || c.peer;
                if(game.playerStates[tid]) c.send({ type: 'gameSync', game: { ...safeGame, myHand: [...game.playerStates[tid].hand] } });
            });
            renderGame({ ...safeGame, myHand: [...game.playerStates[myPeerId].hand] });
            processTurnLogic();
        }

        function handleClientData(data) {
            if (data.type === 'lobbySync') { players = data.players; renderLobby(); }
            else if (data.type === 'gameSync') {
                if(game.status==='waiting' && data.game.status==='playing') switchScreen('screen-game', 'top-status-bar');
                game = { ...game, ...data.game }; renderGame(data.game);
            }
            else if (data.type === 'playSound') playSound(data.name);
            else if (data.type === 'startAnim') doStartAnimation();
            else if (data.type === 'endGame') doEndGame(data.winner, data.scores);
        }

        function doStartAnimation(cb) {
            stopBGM(); playSound('start');
            const animDiv = document.createElement('div'); animDiv.className = 'anim-overlay'; animDiv.style.display = 'flex';
            const text = document.createElement('div'); text.className = 'anim-domino'; text.textContent = 'DOMINO';
            animDiv.appendChild(text); document.body.appendChild(animDiv);
            announce("DOMINO เริ่มเกม!", true);
            setTimeout(() => { animDiv.remove(); if(cb) cb(); }, 3500);
        }

        document.getElementById('btn-start-game').onclick = () => {
            if (!isHost || players.length < 2) return;
            document.getElementById('btn-start-game').style.display = 'none';
            remove(ref(db, `domino_rooms/${currentRoomId}`));
            connections.forEach(c => { if(c.open) c.send({ type: 'startAnim' }); });
            doStartAnimation(() => { initGame(); setTimeout(() => { playSound('bgm'); connections.forEach(c => c.send({ type:'playSound', name:'bgm' })); }, 500); });
        };

        function getTileName(t) {
            if(!t) return '';
            if(t.type === 'number') return `${t.left} กับ ${t.right}`;
            let spec = {'skip':'หลับ', 'draw2':'2+', 'draw3':'3+', 'reverse':'ย้อนศร'};
            return `${t.left} กับ ${spec[t.type]}`;
        }

        function generateDeck() {
            let deck = [];
            for (let i = 0; i <= 9; i++) {
                for (let j = i; j <= 9; j++) deck.push({ type: 'number', left: i, right: j, value: i + j });
            }
            for (let i = 0; i < 5; i++) deck.push({ type: 'skip', left: i, right: 'skip', value: 20 });
            for (let i = 0; i < 5; i++) deck.push({ type: 'draw2', left: i + 5, right: 'draw2', value: 20 });
            for (let i = 0; i < 5; i++) deck.push({ type: 'draw3', left: (i * 2) % 10, right: 'draw3', value: 20 });
            for (let i = 0; i < 5; i++) deck.push({ type: 'reverse', left: (i * 2 + 1) % 10, right: 'reverse', value: 20 });
            return deck.sort(() => Math.random() - 0.5);
        }

        function initGame() {
            game.status = 'playing'; game.deck = generateDeck(); game.log = []; game.direction = 1;
            players.forEach(p => game.playerStates[p.id] = { hand: [], declared: false, playedThisTurn: false });
            players.forEach(p => { for(let i=0; i<7; i++) game.playerStates[p.id].hand.push(game.deck.pop()); });
            
            // Find highest double
            let highest = -1, starterId = players[0].id;
            players.forEach(p => {
                game.playerStates[p.id].hand.forEach(t => {
                    if (t.type === 'number' && t.left === t.right && t.left > highest) { highest = t.left; starterId = p.id; }
                });
            });
            game.turnIndex = players.findIndex(p => p.id === starterId);
            
            let startTile = game.deck.find(t => t.type==='number'); 
            game.leftEnd = startTile.left; game.rightEnd = startTile.right;
            addLog(`เริ่มเกม ปลายคือ ${startTile.left} และ ${startTile.right}`);
            broadcastGameState();
        }

        function isPlayable(tile) {
            if(!game.leftEnd && !game.rightEnd) return true;
            return tile.left === game.leftEnd || tile.left === game.rightEnd || tile.right === game.leftEnd || tile.right === game.rightEnd;
        }

        function addLog(msg) { game.log.unshift(msg); if(game.log.length > 5) game.log.pop(); announce(msg); }

        function drawAuto(playerId) {
            if (game.deck.length === 0) return false;
            playSound('jua'); connections.forEach(c => c.send({type:'playSound', name:'jua'}));
            game.playerStates[playerId].hand.push(game.deck.pop());
            game.playerStates[playerId].declared = false;
            return true;
        }

        function nextTurn() {
            game.playerStates[players[game.turnIndex].id].playedThisTurn = false;
            game.turnIndex = (game.turnIndex + game.direction + players.length) % players.length;
            playSound('turn'); connections.forEach(c => c.send({type:'playSound', name:'turn'}));
            broadcastGameState();
        }

        function handlePlayerAction(peerId, action, payload) {
            if (!isHost || game.status !== 'playing') return;
            const currentPlayer = players[game.turnIndex];
            const state = game.playerStates[peerId];

            if (action === 'call_domino' && state.hand.length === 2 && !state.declared) {
                state.declared = true; addLog(`${players.find(p=>p.id===peerId).name} ประกาศ DOMINO!`);
                playSound('uno'); connections.forEach(c => c.send({type:'playSound', name:'uno'}));
                broadcastGameState(); return;
            }

            if (currentPlayer.id !== peerId) return;

            if (action === 'play') {
                const tile = state.hand[payload.idx];
                state.hand.splice(payload.idx, 1);
                state.playedThisTurn = true;
                
                let matchLeft = (tile.left === game.leftEnd || tile.right === game.leftEnd);
                let matchRight = (tile.left === game.rightEnd || tile.right === game.rightEnd);
                
                let side = payload.side; // 'left' or 'right'
                if (side === 'left') game.leftEnd = (tile.left === game.leftEnd) ? tile.right : tile.left;
                else game.rightEnd = (tile.left === game.rightEnd) ? tile.right : tile.left;

                let tName = getTileName(tile);
                addLog(`${currentPlayer.name} วาง ${tName} ฝั่ง${side==='left'?'ซ้าย':'ขวา'}`);

                if (state.hand.length === 0) { handleWin(peerId); return; }

                if (tile.type === 'skip') { addLog(`ข้ามตาผู้เล่นถัดไป!`); game.turnIndex = (game.turnIndex + game.direction + players.length) % players.length; playSound('skip'); }
                else if (tile.type === 'reverse') { addLog(`ย้อนศร!`); game.direction *= -1; playSound('reverse'); }
                else if (tile.type === 'draw2') { 
                    let nextId = players[(game.turnIndex + game.direction + players.length) % players.length].id;
                    drawAuto(nextId); drawAuto(nextId); addLog(`บังคับจั่ว 2!`); playSound('draw2');
                    game.turnIndex = (game.turnIndex + game.direction + players.length) % players.length;
                }
                else if (tile.type === 'draw3') {
                    let nextId = players[(game.turnIndex + game.direction + players.length) % players.length].id;
                    drawAuto(nextId); drawAuto(nextId); drawAuto(nextId); addLog(`บังคับจั่ว 3!`); playSound('draw4');
                    game.turnIndex = (game.turnIndex + game.direction + players.length) % players.length;
                }

                if (state.hand.some(t => isPlayable(t))) {
                    addLog(`${currentPlayer.name} ได้เล่นต่อในเทิร์นเดียวกัน`);
                    broadcastGameState();
                } else { nextTurn(); }
            } else if (action === 'pass') {
                if (state.playedThisTurn) nextTurn();
            }
        }

        let botTimeout;
        function processTurnLogic() {
            if (!isHost || game.status !== 'playing') return;
            const currentPlayer = players[game.turnIndex];
            const state = game.playerStates[currentPlayer.id];
            
            let validMoves = state.hand.map((t, i) => ({tile: t, idx: i})).filter(x => isPlayable(x.tile));
            
            if (validMoves.length === 0) {
                clearTimeout(botTimeout);
                botTimeout = setTimeout(() => {
                    addLog(`${currentPlayer.name} ไม่มีตัวลง กำลังจั่ว...`);
                    if (drawAuto(currentPlayer.id)) { broadcastGameState(); }
                    else { addLog(`กองจั่วหมด ${currentPlayer.name} ผ่าน`); nextTurn(); }
                }, 1500);
                return;
            }

            if (currentPlayer.isBot) {
                clearTimeout(botTimeout);
                botTimeout = setTimeout(() => {
                    if (state.hand.length === 2 && !state.declared) handlePlayerAction(currentPlayer.id, 'call_domino');
                    let move = validMoves[Math.floor(Math.random() * validMoves.length)];
                    let side = (move.tile.left === game.leftEnd || move.tile.right === game.leftEnd) ? 'left' : 'right';
                    handlePlayerAction(currentPlayer.id, 'play', { idx: move.idx, side });
                }, 2000);
            }
        }

        function handleWin(winnerId) {
            let pts = 0;
            players.forEach(p => { if(p.id !== winnerId) game.playerStates[p.id].hand.forEach(t => pts += t.value); });
            players.find(p=>p.id===winnerId).score += pts;
            game.status = 'ended';
            connections.forEach(c => c.send({ type: 'endGame', winner: players.find(p=>p.id===winnerId).name, scores: players }));
            doEndGame(players.find(p=>p.id===winnerId).name, players);
        }

        function doEndGame(winnerName, pList) {
            stopBGM(); playSound('win');
            switchScreen('screen-result', 'title-result');
            let txt = `🏆 ผู้ชนะรอบนี้: ${winnerName}<br><br>คะแนนสะสม:<br>`;
            pList.forEach(p => txt += `${p.name}: ${p.score} คะแนน<br>`);
            document.getElementById('winner-text').innerHTML = txt;
            if(isHost) {
                document.getElementById('result-host-controls').style.display = 'flex';
                document.getElementById('btn-play-again').onclick = () => { document.getElementById('btn-start-game').click(); };
            }
        }

        let pendingPlayIdx = -1;
        function renderGame(gameState) {
            const myState = gameState.playerStates[myPeerId];
            const isMyTurn = players[gameState.turnIndex].id === myPeerId;
            
            document.getElementById('current-turn-heading').textContent = `รอบของ ${players[gameState.turnIndex].name}`;
            if(gameState.log.length>0) document.getElementById('event-log').textContent = gameState.log[0];
            
            document.getElementById('deck-count-visual').textContent = gameState.deckCount;
            document.getElementById('left-end-container').innerHTML = `<div class="domino-tile"><div class="half">${gameState.leftEnd!==null?gameState.leftEnd:''}</div></div>`;
            document.getElementById('right-end-container').innerHTML = `<div class="domino-tile"><div class="half">${gameState.rightEnd!==null?gameState.rightEnd:''}</div></div>`;

            const handDiv = document.getElementById('my-cards-container'); handDiv.innerHTML = '';
            (gameState.myHand || []).forEach((tile, idx) => {
                let playable = isMyTurn && (tile.left === gameState.leftEnd || tile.right === gameState.leftEnd || tile.left === gameState.rightEnd || tile.right === gameState.rightEnd);
                let btn = document.createElement('button');
                btn.className = `domino-tile ${!playable ? 'disabled' : ''} ${tile.type !== 'number' ? 'special' : ''}`;
                btn.innerHTML = `<div class="half">${tile.left}</div><div class="divider"></div><div class="half">${tile.type==='number'?tile.right: `<span class="card-special-text">${getTileName(tile).split(' ')[2]}</span>`}</div>`;
                if(playable) {
                    btn.onclick = () => {
                        let mLeft = tile.left === gameState.leftEnd || tile.right === gameState.leftEnd;
                        let mRight = tile.left === gameState.rightEnd || tile.right === gameState.rightEnd;
                        if(mLeft && mRight && gameState.leftEnd !== gameState.rightEnd) {
                            pendingPlayIdx = idx; document.getElementById('choice-modal').style.display = 'flex';
                        } else {
                            handlePlayerAction(myPeerId, 'play', { idx, side: mLeft ? 'left' : 'right' });
                        }
                    };
                }
                handDiv.appendChild(btn);
            });

            document.getElementById('btn-pass').disabled = !(isMyTurn && myState.playedThisTurn);
            document.getElementById('btn-pass').onclick = () => handlePlayerAction(myPeerId, 'pass');
            document.getElementById('btn-domino-call').disabled = !(isMyTurn && gameState.myHand.length === 2 && !myState.declared);
            document.getElementById('btn-domino-call').onclick = () => handlePlayerAction(myPeerId, 'call_domino');

            document.getElementById('top-status-bar').innerHTML = '';
            players.forEach((p, i) => {
                let st = gameState.playerStates[p.id];
                if(st) {
                    let div = document.createElement('div'); div.className = `character-card ${i===gameState.turnIndex?'is-turn':''}`;
                    div.innerHTML = `<div class="char-cards-count">${st.cardCount} ตัว</div><div class="char-avatar">${p.isBot?'🤖':'👤'}</div><div class="char-score">${p.score}</div>`;
                    document.getElementById('top-status-bar').appendChild(div);
                }
            });
        }

        document.getElementById('btn-choice-left').onclick = () => { document.getElementById('choice-modal').style.display='none'; handlePlayerAction(myPeerId, 'play', { idx: pendingPlayIdx, side: 'left' }); pendingPlayIdx = -1; };
        document.getElementById('btn-choice-right').onclick = () => { document.getElementById('choice-modal').style.display='none'; handlePlayerAction(myPeerId, 'play', { idx: pendingPlayIdx, side: 'right' }); pendingPlayIdx = -1; };
        document.getElementById('btn-choice-cancel').onclick = () => { document.getElementById('choice-modal').style.display='none'; pendingPlayIdx = -1; };

    </script>
